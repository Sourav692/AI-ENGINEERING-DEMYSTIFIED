Validate deterministically first, then judge

# The pattern from the notebook: a hard schema check runs BEFORE the LLM metric.
# A structurally malformed call is worth zero — no point spending a judge call
# to have a model tell you the date field isn't a date.

TOOL_SCHEMAS = {
    "FlightSearch": {
        "required_keys": {"origin", "destination", "date"},
        "validators": {"date": (lambda v: datetime.strptime(v, "%Y-%m-%d"), "YYYY-MM-DD")},
    }
}

def validate_args(tools_called) -> tuple[bool, str]:
    for tool in tools_called:
        schema = TOOL_SCHEMAS.get(tool.name)
        if not schema: continue
        actual = set(tool.input_parameters.keys())
        missing, extra = schema["required_keys"] - actual, actual - schema["required_keys"]
        if missing: return False, f"{tool.name}: missing {missing}"
        if extra:   return False, f"{tool.name}: unexpected {extra}"
        for field, (validator, fmt) in schema["validators"].items():
            try: validator(tool.input_parameters.get(field))
            except (ValueError, TypeError):
                return False, f"{tool.name}: '{field}' must be {fmt}"
    return True, "ok"

valid, reason = validate_args(test_case.tools_called)
if not valid:
    print(f"Score: 0\nReason: {reason}")        # judge never runs — free rejection
else:
    metric.measure(test_case)                    # now spend the LLM call
